-- Table Users -- 

1. Path : /hms/user/create 
    Method : POST 
    Parameters : <Parameters for creation. Refer: Columns in the table> 
    Description : Add new user to database
    Role: ADMIN

2. Path : /hms/user/lookup
    Method : GET
    Parameters : Filter: User_ID or Phone or Name 
    Description : View user details and profile (Doctor, Patient)
    Role: ADMIN, DOCTOR, PATIENT

3. Path : /hms/user/update
    Method : POST 
    Parameters : <Key: User_ID, key-value pair>
    Description : Edit user details and profile
    Role : ADMIN, DOCTOR, PATIENT

4. Path : /hms/user/delete/
    Method : DELETE 
    Parameters : User_ID 
    Description : Deletes user (and by extension appointments and history?) from database 
    Role : ADMIN 

-- Table Appointments --

1. Path : /hms/appointment/create
    Method : POST
    Parameters : Patient_ID + other columns in the appointment entry
    Description : User [Patient] can schedule appointments 
    Role : ADMIN, PATIENT

2. Path : /hms/appointments/lookup 
    Method : GET
    Parameters : Filter : Patient_ID or Doctor_ID or Appointment_ID or Appointment_Status
    Description : Page displays all appointments, specific appointments via ID or appointments specific to a user 
                  Admins can also view previous and upcoming appointments
    Role : ADMIN, DOCTOR, PATIENT

3. Path : /hms/appointment/update
    Method : POST 
    Parameters : < Key : Appointment_ID or Patiend_ID or Doctor_ID >
    Description : User can update appointment status [Only Scheduled or Completed]
    Role : ADMIN, DOCTOR

4. Path : /hms/appointments/delete
    Method : DELETE 
    Parameters : Appointment_ID or Patient_ID 
    Description : User (patient) can cancel appointments 
    Role : ADMIN, PATIENT
