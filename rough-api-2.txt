-- Table Users -- 

1. Path : /hms/user/create 
    Method : POST 
    Parameters : 
    Object = {
        "User_ID" : <user_id>, "First_name" : <fname>, "Last_name" : <lname> , "User_Type" : <enum ADMIN,PATIENT OR DOCTOR>,
        "Phone_Number" : <ph_num> , "User_Profile" : <Text> , "Password" : <pswd>
    }
    Description : Add new user to database
    Role: ADMIN

2. Path : /hms/user/lookup
    Method : GET
    Parameters : 
    Object = {
        "User_ID" : <user_id>, "Phone_Number" : <ph_num>, "First_name" : <fname>, "Last_name" : <lname>
    }
    Description : View user details and profile (Doctor, Patient)
    Role: ADMIN, DOCTOR, PATIENT

3. Path : /hms/user/update
    Method : POST 
    Parameters :
    Object = {
        "User_ID" = <user_id>
    }
    Description : Edit user details and profile
    Role : ADMIN, DOCTOR, PATIENT

4. Path : /hms/user/delete/
    Method : DELETE 
    Parameters : 
    Object = {
        "User_ID" = <user_id>, "First_name" : <fname>, "Last_name" : <lname> , "User_Type" : <enum ADMIN,PATIENT OR DOCTOR>,
        "Phone_Number" : <ph_num> , "User_Profile" : <Text> , "Password" : <pswd>
    }
    Description : Deletes user (and by extension appointments and history?) from database 
    Role : ADMIN
